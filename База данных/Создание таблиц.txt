-- 1. Справочник производственных участков
CREATE TABLE work_areas (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    sort_order INT DEFAULT 0
);

-- 2. Справочник оборудования
CREATE TABLE equipment (
    id SERIAL PRIMARY KEY,
    work_area_id INT NOT NULL REFERENCES work_areas(id),
    name VARCHAR(255) NOT NULL,
    code VARCHAR(50) UNIQUE, -- Инвентарный номер / Шифр
    sort_order INT DEFAULT 0
);

-- 3. Справочник должностей
CREATE TABLE positions (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL -- 'Токарь', 'Мастер', 'Начальник цеха'
);

-- 4. Мастер-таблица сотрудников
CREATE TABLE employees (
    id SERIAL PRIMARY KEY,
    full_name VARCHAR(255) NOT NULL,
    system_role ENUM('worker', 'master', 'chief') DEFAULT 'worker' NOT NULL
);

-- 5. Периоды работы (Прием / Увольнение)
-- Позволяет учитывать повторные наймы одного и того же человека
CREATE TABLE employment_periods (
    id SERIAL PRIMARY KEY,
    employee_id INT NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
    hire_date DATE NOT NULL,
    fire_date DATE NULL, -- Если NULL, сотрудник работает по настоящее время
    CONSTRAINT check_employment_dates CHECK (fire_date IS NULL OR fire_date >= hire_date)
);

-- 6. История должностей
CREATE TABLE employee_position_assignments (
    id SERIAL PRIMARY KEY,
    employee_id INT NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
    position_id INT NOT NULL REFERENCES positions(id),
    valid_from DATE NOT NULL,
    UNIQUE(employee_id, valid_from)
);

-- 7. История закрепления основного оборудования
CREATE TABLE employee_equipment_assignments (
    id SERIAL PRIMARY KEY,
    employee_id INT NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
    equipment_id INT NULL REFERENCES equipment(id), -- NULL для должностей без привязки
    valid_from DATE NOT NULL,
    UNIQUE(employee_id, valid_from)
);

-- 8. Справочник смен
CREATE TABLE shift_definitions (
    id SERIAL PRIMARY KEY,
    shift_number INT NOT NULL, -- 1, 2, 3, 4 или 0 (выходной)
    name VARCHAR(50) NOT NULL,  -- Утро, Вечер, Ночь, Выходной
    start_time TIME,
    end_time TIME,
    is_work_day BOOLEAN DEFAULT TRUE
);

-- 9. Шаблоны графиков (Бригады)
CREATE TABLE schedule_templates (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL, -- 'Бригада А', 'Офис 5/2'
    base_date DATE NOT NULL,      -- Точка отсчета цикла
    cycle_length INT NOT NULL     -- Длина цикла в днях (напр. 28 для 4 недель)
);

-- 10. Дни внутри цикла графиков (состав шаблона)
CREATE TABLE schedule_items (
    template_id INT NOT NULL REFERENCES schedule_templates(id) ON DELETE CASCADE,
    day_number INT NOT NULL,      -- Порядковый день в цикле
    shift_id INT NOT NULL REFERENCES shift_definitions(id),
    PRIMARY KEY (template_id, day_number)
);

-- 11. История переходов сотрудников между бригадами
CREATE TABLE employee_schedule_assignments (
    id SERIAL PRIMARY KEY,
    employee_id INT NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
    template_id INT NOT NULL REFERENCES schedule_templates(id),
    valid_from DATE NOT NULL,
    UNIQUE(employee_id, valid_from)
);

-- 12. Справочник типов отсутствий
CREATE TABLE absence_types (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL -- 'Отпуск', 'Больничный', 'Отгул'
);

-- 13. Журнал отсутствий (с поддержкой открытых дат)
CREATE TABLE absences (
    id SERIAL PRIMARY KEY,
    employee_id INT NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
    type_id INT NOT NULL REFERENCES absence_types(id),
    start_date DATE NOT NULL,
    end_date DATE NULL, -- Если NULL, отсутствие считается открытым до текущего момента
    CONSTRAINT check_absence_dates CHECK (end_date IS NULL OR end_date >= start_date)
);

-- 14. Журнал ручных правок и Workflow согласования
CREATE TABLE schedule_overrides (
    id SERIAL PRIMARY KEY,
    employee_id INT NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
    override_date DATE NOT NULL,
    shift_id INT REFERENCES shift_definitions(id),      -- Новая смена
    equipment_id INT REFERENCES equipment(id),          -- Смена оборудования на день
    status INT DEFAULT 0,                               -- 0:Черновик, 1:Подтверждено, 2:Утверждено, -1:Отмена
    comment TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(employee_id, override_date)
);

-- 15. Дополнительные участки доступа (многие-ко-многим)
CREATE TABLE employee_areas (
    employee_id INT NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
    work_area_id INT NOT NULL REFERENCES work_areas(id) ON DELETE CASCADE,
    PRIMARY KEY (employee_id, work_area_id)
);
